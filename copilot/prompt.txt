import logging, requests, json
from datetime import datetime, timedelta
from requests.adapters import HTTPAdapter
from urllib3.util.retry import Retry
from django.conf import settings
from channels.layers import get_channel_layer
from asgiref.sync import async_to_sync
from celery import shared_task
from polygon import RESTClient
from data_factory import engine

logger = logging.getLogger(__name__)

# Configure retry strategy for API calls

    total=3,
    backoff_factor=1,
    status_forcelist=[429, 500, 502, 503, 504],
)
adapter = HTTPAdapter(max_retries=retry_strategy)
http = requests.Session()
http.mount("https://", adapter)
http.mount("http://", adapter)

@shared_task(bind=True, max_retries=3)
def fetch_polygon_data(self, asset_class: str, single_symbol: str=None):
    """
    Fetch time-series data from Polygon for various asset classes,
    process OHLCV data, generate market intelligence narratives.
    """
    API_KEY = getattr(settings, "POLYGON_API_KEY", None)
    if not API_KEY:
        logger.error("POLYGON_API_KEY not set in settings.")
        return None

    # Configurable defaults based on asset class
    if asset_class == 'forex':
        symbols = json.loads(getattr(settings, "POLYGON_FX_PAIRS", '["EURUSD","GBPUSD","USDJPY","AUDUSD"]'))
        timespan = getattr(settings, "POLYGON_FX_TIMESPAN", "day")
        limit = getattr(settings, "POLYGON_FX_LIMIT", 50)
        ticker_prefix = "C:"
    elif asset_class == 'stocks':
        symbols = json.loads(getattr(settings, "POLYGON_STOCK_SYMBOLS", '["AAPL","MSFT","GOOGL","AMZN"]'))
        timespan = getattr(settings, "POLYGON_STOCK_TIMESPAN", "day")
        limit = getattr(settings, "POLYGON_STOCK_LIMIT", 50)
        ticker_prefix = ""
    elif asset_class == 'crypto':
        symbols = json.loads(getattr(settings, "POLYGON_CRYPTO_SYMBOLS", '["BTCUSD","ETHUSD","XRPUSD"]'))
        timespan = getattr(settings, "POLYGON_CRYPTO_TIMESPAN", "day")
        limit = getattr(settings, "POLYGON_CRYPTO_LIMIT", 50)
        ticker_prefix = "X:"
    else:
        logger.error(f"Unsupported asset class: {asset_class}")
        return None

    # Dynamic dates - get more data for better analysis
    end_date = datetime.utcnow().strftime("%Y-%m-%d")
    start_date = (datetime.utcnow() - timedelta(days=30)).strftime("%Y-%m-%d")

    client = RESTClient(API_KEY)
    raw_data = {}

    for symbol in symbols:
        clean_symbol = symbol.upper().replace('/', '')
        ticker = f"{ticker_prefix}{clean_symbol}"
        aggs = []
        
        try:
            resp = client.list_aggs(
                ticker=ticker,
                multiplier=1,
                timespan=timespan,
                from_=start_date,
                to=end_date,
                limit=limit,
            )
            
            for bar in resp:
                aggs.append({
                    "timestamp": bar.timestamp,
                    "open": bar.open,
                    "high": bar.high,
                    "low": bar.low,
                    "close": bar.close,
                    "volume": bar.volume,
                })
            
            raw_data[symbol] = aggs
            logger.info(f"Successfully fetched {asset_class} data for {symbol}")
            
        except Exception as e:
            logger.error(f"Error fetching {asset_class} data for {ticker}: {e}")
            # Retry the task if there's an error
            try:
                self.retry(exc=e, countdown=60)
            except self.MaxRetriesExceededError:
                logger.error(f"Max retries exceeded for {asset_class} data fetch")
            raw_data[symbol] = []


    
    formatted_data = engine.format_asset_data(raw_data, symbols)
    
    # If a single symbol is requested, return only its overview
    if single_symbol:
        try:
            forex_overview = engine.get_forex_asset_overview(formatted_data, single_symbol)
            logger.debug(f"fx overview >>>>>>>> : {forex_overview}")
            logger.info(f"Processed single symbol {single_symbol} for {asset_class}")
            
            # Broadcast to appropriate WebSocket channel
            try:
                channel_layer = get_channel_layer()
                async_to_sync(channel_layer.group_send)(
                    f"symbol_intelligence_{asset_class}_{single_symbol}",
                    {
                        "type": "symbol.intelligence",
                        "message": forex_overview
                    }
                )
                logger.info(f"Symbol intelligence broadcasted for {single_symbol}")
            except Exception as e:
                logger.error(f"Failed to broadcast symbol intelligence: {e}")
                
            
        except Exception as e:
            logger.error(f"Error processing single symbol {single_symbol}: {e}")
    
    # Otherwise, process the full market data
    try:
        processed_data = engine.process_asset_data(
            formatted_data, 
            asset_class, 
            breadth_series_len=20, 
            top_n=5
        )
        
        # Add metadata
        processed_data.update({
            'asset_class': asset_class,
            'generated_at': datetime.utcnow().isoformat(),
            'symbols': symbols
        })
        
        logger.info(f"Processed {asset_class} data with {len(processed_data.get('narratives', []))} narratives")
        
        # Broadcast to appropriate WebSocket channel
        try:
            channel_layer = get_channel_layer()
            async_to_sync(channel_layer.group_send)(
                f"market_intelligence_{asset_class}",
                {
                    "type": "market.intelligence",
                    "message": processed_data
                }
            )
            logger.info(f"Market intelligence broadcasted to {asset_class} WebSocket channel")
        except Exception as e:
            logger.error(f"Failed to broadcast to WebSocket: {e}")
            
        return processed_data, forex_overview
        
    except Exception as e:
        logger.error(f"Error processing {asset_class} data: {e}")
        # Create a basic response with error information
        return {
            'asset_class': asset_class,
            'generated_at': datetime.utcnow().isoformat(),
            'error': str(e),
            'symbols': symbols,
            'data': raw_data
        }

This enhanced dashboard provides a comprehensive market overview while allowing users to drill down into specific pairs for more detailed analysis, all without changing the original styling or adding buy/sell signals.



1. Overall Market Health & Status

These metrics give a top-level view of market sentiment and conditions.

    Market Status (Bullish/Bearish/Neutral):

        How to Calculate: A composite score based on multiple indicators below (e.g., Advancers/Decliners, VIX level, Index performance). A simple rule could be: If >55% of S&P 500 stocks are above their 50-day MA, status is "Bullish"; if <45%, "Bearish"; else "Neutral".

    Market Breadth (Advancers vs. Decliners):

        Metric: Advancing Issues / Total Issues for a major index like the S&P 500 or NYSE.

        Data Needed: Number of stocks that closed up (advancers) and down (decliners) on the primary exchange.

    Volatility Index (VIX):

        Metric: The CBOE Volatility Index value.

        How to Get: This is a direct data point provided by most APIs. You don't calculate it yourself; you fetch it.

    Total Market Volume vs. Average:

        Metric: Total Volume Today / Average Daily Volume (e.g., 30-day ADTV).

        Data Needed: Cumulative volume of all stocks on an exchange or index and their 30-day average volume.

2. Index & Sector Performance

These metrics track the performance of major market segments.

    Index Prices & Returns:

        Metrics: Raw price, daily change ((Current Price - Previous Close) / Previous Close), weekly change, YTD change for:

            S&P 500 (SPX, SPY)

            Nasdaq Composite (IXIC) or Nasdaq-100 (NDX, QQQ)

            Dow Jones Industrial Average (DJI, DIA)

            Russell 2000 (RUT, IWM)

    Sector Performance:

        Metrics: Daily, weekly, monthly, and YTD returns for key sectors (Technology, Healthcare, Financials, Energy, etc.).

        How to Get: Typically calculated using Sector ETFs (e.g., XLK for Tech, XLV for Health, XLF for Financials) or by aggregating the performance of all stocks within a sector classification.

3. Individual Stock Metrics

These are the core fundamentals and technicals for each stock in your overview table.

    Price & Performance:

        Current Price

        Previous Close

        Daily Change ($): Current Price - Previous Close

        Daily Change (%): (Daily Change $ / Previous Close) * 100

        High & Low (for the day)

    Volume & Liquidity:

        Volume: Shares traded today.

        Average Volume (30-Day): The average daily volume over the past 30 trading days.

    Valuation Metrics:

        Market Capitalization: Current Price * Total Shares Outstanding

        Price-to-Earnings Ratio (P/E): Current Price / Earnings Per Share (EPS) (Usually Trailing Twelve Months - TTM)

    Technical Indicators (Crucial for the "Trend" column and strength gauges):

        Simple Moving Average (SMA) Position: (Current Price / SMA(50)) - 1 to see if price is above/below its 50-day average. This powers the "Bullish Instruments" donut chart.

        Trend Strength: Can be derived from the slope of a moving average or an oscillator like the Average Directional Index (ADX). A value above 25 often indicates a strong trend.

        52-Week High/Low: (Current Price - 52W Low) / (52W High - 52W Low) * 100 (This gives a % through the yearly range).

4. Market Internals & Correlations

These are more advanced metrics that describe the relationships and underlying forces in the market.

    Correlation Coefficients:

        Metric: A statistical measure (from -1 to +1) of how two indices or assets move in relation to each other.

        Example: Correlation between SPY (S&P 500 ETF) and QQQ (Nasdaq-100 ETF). Calculated over a rolling window (e.g., 20 days).

    New Highs vs. New Lows: The number of stocks hitting their 52-week high versus those hitting their 52-week low. A strong breadth indicator.

    Put/Call Ratio: The ratio of trading volume in put options to call options. A high ratio can indicate fear (bearish sentiment), while a low ratio can indicate greed (bullish sentiment). Fetched from an options data API.

5. Top Movers

    Gainers: Stocks with the highest positive Daily Change (%).

    Losers: Stocks with the highest negative Daily Change (%).

    Most Active: Stocks with the highest absolute Volume.
